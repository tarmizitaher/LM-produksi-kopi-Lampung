\section{Konsep Dasar Kinetika Adsorpsi}
\label{sec:konsep}

Kapasitas adsorpsi pada waktu $t$ dihitung dari neraca massa:
\begin{equation}
  q_t = \frac{(C_0 - C_t)\,V}{m},
  \label{eq:qt}
\end{equation}
dengan $q_t$ jumlah adsorbat teradsorpsi (\si{\milli\gram\per\gram}),
$C_0$ dan $C_t$ konsentrasi awal dan pada waktu $t$
(\si{\milli\gram\per\liter}), $V$ volume larutan (\si{\liter}), dan
$m$ massa adsorben (\si{\gram}).

(ii) difusi intrapartikel, (iii) adsorpsi pada situs aktif. Tahap

\section{Model Reaksi Permukaan}
\label{sec:reaksi}

\subsection{Pseudo-orde satu (PFO)}
Model Lagergren mengasumsikan laju sebanding dengan jumlah situs kosong:
\begin{equation}
  \frac{dq_t}{dt} = k_1 (q_e - q_t).
  \label{eq:pfo-diff}
\end{equation}
Integrasi dengan $q_t = 0$ pada $t = 0$ menghasilkan
\begin{equation}
  q_t = q_e\left(1 - e^{-k_1 t}\right),
  \label{eq:pfo}
\end{equation}
dengan bentuk linear $\ln(q_e - q_t) = \ln q_e - k_1 t$.
\todo{Sitasi sumber asli Lagergren (1898) --- belum terverifikasi.}

\subsection{Pseudo-orde dua (PSO)}
\begin{equation}
  \frac{dq_t}{dt} = k_2 (q_e - q_t)^2
  \quad\Longrightarrow\quad
  q_t = \frac{k_2 q_e^2\, t}{1 + k_2 q_e\, t},
  \label{eq:pso}
\end{equation}
dengan bentuk linear yang paling sering dipakai
\begin{equation}
  \frac{t}{q_t} = \frac{1}{k_2 q_e^2} + \frac{t}{q_e},
  \label{eq:pso-lin}
\end{equation}
serta laju adsorpsi awal $h = k_2 q_e^2$
\parencite{Ho1999_PseudoSecondOrder}.

\todo{Diskusi kritis: mengapa PSO hampir selalu ``menang''
\parencite{Simonin2016_PFOvsPSO}.}

\subsection{Model Elovich}
\begin{equation}
  \frac{dq_t}{dt} = \alpha\, e^{-\beta q_t}
  \quad\Longrightarrow\quad
  q_t = \frac{1}{\beta}\ln(1 + \alpha\beta t),
  \label{eq:elovich}
\end{equation}
dengan $\alpha$ laju adsorpsi awal dan $\beta$ konstanta desorpsi.
\todo{Sitasi dan interpretasi (permukaan heterogen, kemisorpsi).}

\begin{equation}
  q_t = q_e\left\{1 - \exp\!\left[-(k_{AV}\, t)^{n_{AV}}\right]\right\}.

\section{Model Difusi}
\label{sec:difusi}

\subsection{Difusi intrapartikel (Weber--Morris)}
\begin{equation}
  q_t = k_{id}\, t^{1/2} + C,
  \label{eq:weber-morris}
\end{equation}
dengan $C$ terkait ketebalan lapisan batas. Plot multilinear
menunjukkan lebih dari satu tahap pengendali laju.
\todo{Sitasi Weber \& Morris (1963) --- belum terverifikasi.}

\subsection{Model film Boyd}
Dengan $F = q_t/q_e$,
\begin{equation}
  F = 1 - \frac{6}{\pi^2}\sum_{n=1}^{\infty}\frac{1}{n^2}
      \exp\!\left(-n^2 B t\right),
  \label{eq:boyd}
\end{equation}
yang didekati dengan
\begin{equation}
  Bt =
  \begin{cases}
    \left(\sqrt{\pi} - \sqrt{\pi - \dfrac{\pi^2 F}{3}}\right)^{2},
      & F \le 0.85,\\[2ex]
    -0.4977 - \ln(1 - F), & F > 0.85.
  \end{cases}
  \label{eq:boyd-approx}
\end{equation}
Garis $Bt$ terhadap $t$ yang tidak melalui titik asal menandakan difusi
film ikut mengendalikan laju.
\todo{Sitasi Boyd dkk.\ (1947) --- belum terverifikasi.}

\section{Estimasi Parameter dan Pemilihan Model}
\label{sec:estimasi}

\todo{Linear vs non-linear: transformasi linear mengubah struktur galat
dan membiaskan parameter
\parencite{Foo2010_IsothermModeling,Tran2017_AdsorptionMistakes}.}

Beberapa fungsi galat yang umum dipakai untuk membandingkan model:
\begin{align}
  \chi^2 &= \sum_{i=1}^{N}
            \frac{(q_{e,\mathrm{exp}} - q_{e,\mathrm{cal}})_i^2}
                 {q_{e,\mathrm{cal},i}},
  \label{eq:chi2}\\
  \mathrm{RMSE} &= \sqrt{\frac{1}{N}\sum_{i=1}^{N}
            (q_{\mathrm{exp}} - q_{\mathrm{cal}})_i^2},
  \label{eq:rmse}\\
  \mathrm{AIC} &= N \ln\!\left(\frac{\mathrm{SSE}}{N}\right) + 2p,
  \label{eq:aic}
\end{align}
dengan $N$ jumlah titik data, SSE jumlah kuadrat galat, dan $p$ jumlah
parameter model.
